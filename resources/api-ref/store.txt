## store — browse and install from the app store
covers: browse and search store listings, read a listing, install and
        uninstall store items, install from a local .dhpkg file, skills
not here:
  configuring an installed digital human -> digital-human
  knowledge collections -> knowledge-base
  installing or removing a skill -> use the skill_manage tool

POST   /api/apps/<appId>/upgrade-strategy                          Set how a store-installed digital human handles version upgrades
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/apps/<appId>/upgrade-strategy" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"strategy":"auto"}'
     # body: {"strategy": "auto"}
     # out: {success:true}
     # note: strategy is one of "auto" (silent patch/minor, notify on major), "notify" (always notify), or "manual" (no automatic checks)

GET    /api/skills/command-name                                    Preview the slash-command name a skill title would install under
     curl -s "$HALO_API_URL/api/skills/command-name" -H "Authorization: Bearer $HALO_API_TOKEN"
     # out: {success:true,data:string}  // e.g. "My Skill" -> "my-skill"
     # note: Query: ?name=. A missing name returns an empty string, not a 400. Pure string transform — it does not check whether that name is already taken.

POST   /api/skills/copy                                            Create an independent personal skill copy under a new name
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/skills/copy" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"ref":{"id":"<copy.id>","spaceId":"$HALO_SPACE_ID"},"spaceId":"$HALO_SPACE_ID","name":"my-copy"}'
     # body: {"ref":{"id":"<copy.id>","spaceId":"$HALO_SPACE_ID"},"spaceId":"$HALO_SPACE_ID","name":"my-copy"}
     # out: {success:true,data:<SkillCopy>} or {success:false,error}
     # note: Destination spaceId is required and used exactly as supplied; null means global. name is required and at most 128 characters; service filename/command validation also applies. Optional content replaces the copied SKILL.md (at most 1 MiB UTF-8); supporting files are copied with the package.

POST   /api/skills/delete                                          Trash one skill package on the host computer
     !! DESTRUCTIVE: cannot be undone. Confirm with the user in plain language before running this.
     !! If the user meant something smaller, use /api/skills/set-enabled instead.
     curl -sX POST "$HALO_API_URL/api/skills/delete" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"ref":{"id":"<copy.id>","spaceId":"$HALO_SPACE_ID"},"revision":"<revision from read>"}'
     # body: {"ref":{"id":"<copy.id>","spaceId":"$HALO_SPACE_ID"},"revision":"<revision from read>"}
     # out: {success:true} or {success:false,error}
     # note: Requires the current revision and deletes only this copy. The host computer recycle bin/trash receives the files, not the remote client. There is no API undo; disabling instead preserves all files. A lower-priority copy may become effective.

POST   /api/skills/import                                          Import a complete personal skill package into an explicit scope
     !! DESTRUCTIVE: cannot be undone. Confirm with the user in plain language before running this.
     curl -sX POST "$HALO_API_URL/api/skills/import" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"spaceId":"$HALO_SPACE_ID","requestId":"<stable request ID>","spec":{"type":"skill","name":"my-skill","version":"1.0","description":"What it does","skill_content":"<complete SKILL.md>"}}'
     # body: {"spaceId":"$HALO_SPACE_ID","requestId":"<stable request ID>","spec":{"type":"skill","name":"my-skill","version":"1.0","description":"What it does","skill_content":"<complete SKILL.md>"}}
     # out: {success:true,data:<SkillCopy>} or {success:false,error}
     # note: spaceId is required and used exactly as supplied. null means global; another workspace ID writes there. Only SkillSpec is accepted; importing a digital human uses its dedicated creation tool instead.
     # note: requestId is required: 8–128 ASCII letters, digits, underscores or hyphens. Reuse it only when retrying the same import. Supply package bytes, never a client path or arbitrary host filePath.
     # note: Multi-file specs use skill_files for relative UTF-8 resources and skill_assets for {encoding:"base64",content:"..."} binary resources. Service validation owns path containment and package limits; HTTP body limits also apply.
     # note: An existing copy is not implicitly overwritten. Explicit replacement adds replace:{ref:{id,spaceId},revision} using a fresh read; only a personal copy in the target scope can be replaced. Replacement removes obsolete package resources and preserves its disabled choice.

POST   /api/skills/list                                            List skill copies available in one workspace and global scope
     curl -sX POST "$HALO_API_URL/api/skills/list" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"spaceId":"$HALO_SPACE_ID"}'
     # body: {"spaceId":"$HALO_SPACE_ID"}
     # out: {success:true,data:{groups:[{name,displayName,description,source,copies:[{id,appId?,name,displayName,description,path,scope,spaceId,source,owners,enabled,availability,error?,revision,editable,isLink}],effectiveId?,conflict}],offset,limit,total,truncated,warnings,legacyRemoved,engines}} or {success:false,error}
     # note: spaceId is required: a workspace ID includes that workspace and global copies; null inspects only global scope. The supplied workspace is used exactly as sent, including another workspace.
     # note: Optional q (at most 1024 characters), source (all|mine|digital-human|builtin), offset (non-negative safe integer), limit (1–100; default 50). Follow pagination and report warnings/truncation rather than treating uninspected files as absent.
     # note: Copy IDs are opaque discovered references, not paths. A ref is {id:copy.id,spaceId:<the inspected workspace or null>}; read again before file mutations. Do not invent IDs or use a path as an ID.

POST   /api/skills/read                                            Read one discovered skill copy and its current file revision
     curl -sX POST "$HALO_API_URL/api/skills/read" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"id":"<copy.id from POST /api/skills/list>","spaceId":"$HALO_SPACE_ID"}'
     # body: {"id":"<copy.id from POST /api/skills/list>","spaceId":"$HALO_SPACE_ID"}
     # out: {success:true,data:{copy,content,revision,files:[{path,size,binary}],spec}} or {success:false,error}
     # note: Send the ref directly, not inside a ref field. IDs accept file:<64 lowercase hex digits> or app:<installation UUID> for legacy navigation; the service resolves them again within the inspected workspace and global roots. No arbitrary filePath is accepted.

POST   /api/store/apps/<slug>/install                              Install an app from the store by slug (REST-style path variant)
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/store/apps/<slug>/install" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"spaceId":"$HALO_SPACE_ID"}'
     # body: {"spaceId":"$HALO_SPACE_ID"}
     # out: {success:true,data:{appId}} or {success:false,error,code?}
     # note: Always send spaceId, and send the right one. It is used exactly as sent: nothing fills it in and nothing checks it against the space you are working in. Omit it and the install goes GLOBAL; name another space and it installs there. Both answer success:true, and the response never says which space was written.
     # note: Pass "spaceId":null only when the user explicitly asked for a global install (mcp/skill types only).
     # note: Optional field: userConfig (object).
     # note: Identical behavior to POST /api/store/install (slug in path instead of body), including the 403 MCP_COMMAND_BLOCKED case.

POST   /api/store/import-dhpkg                                     Install a digital human from a local .dhpkg file

          -H "Authorization: Bearer $HALO_API_TOKEN" \

     # note: spaceId is used exactly as sent — omitted means a global install, another space means it installs there, and neither is reported back. 400 when filePath is missing.
     # note: Activation failing after install still returns success:true; check GET /api/apps/<appId> to confirm it is running.

     # out: {success:true,data:{id,name,url,enabled,isDefault:false,sourceType?,adapterConfig?:{...,"apiKey":"[redacted]"}}} or {success:false,error}
     # note: Optional fields: sourceType (string), adapterConfig (object, adapter-specific — e.g. an API key).
     # note: url must be http(s) and not point to a loopback/private/link-local host. Fails if a registry with the same URL already exists.

DELETE /api/store/registries/<registryId>                          Remove a configured registry source
     # note: reversible — can be undone or corrected afterward.
     curl -sX DELETE "$HALO_API_URL/api/store/registries/<registryId>" -H "Authorization: Bearer $HALO_API_TOKEN"
     # out: {success:true,data:undefined} or {success:false,error}
     # note: Fails with an error (not thrown) when the target is the built-in/default registry — those cannot be removed. Can be added back with POST /api/store/registries.

PATCH  /api/store/registries/<registryId>/adapter-config           Update a registry source's adapter config (e.g. its API key)
     # note: reversible — can be undone or corrected afterward.
     curl -sX PATCH "$HALO_API_URL/api/store/registries/<registryId>/adapter-config" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"apiKey":"sk-..."}'
     # body: {"apiKey":"sk-..."}
     # out: {success:true,data:undefined}
     # note: Body is a flat JSON object of adapter-specific fields — exact keys depend on the registry's sourceType.
     # note: Write-only: the response never echoes the config back, so this is safe for setting credentials. 400 if the body is missing or not a plain object.

POST   /api/store/registries/<registryId>/toggle                   Enable or disable a configured registry source
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/store/registries/<registryId>/toggle" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"enabled":true}'
     # body: {"enabled":true}
     # out: {success:true,data:undefined}
     # err: 400 if enabled is missing or not a boolean.

POST   /api/store/revalidate                                       Cheap conditional freshness check for the store index
     curl -sX POST "$HALO_API_URL/api/store/revalidate" -H "Authorization: Bearer $HALO_API_TOKEN"
     # out: {success:true,data:{changed:boolean}}
     # note: changed:false means the index did not move (a 304-equivalent); the browse list does not need refetching.

POST   /api/store/unpublish                                        Take down one of the creator's own published apps
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/store/unpublish" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"slug":"author/app-id"}'
     # body: {"slug":"author/app-id"}
     # out: {success:true,data:null}
     # note: Can be republished later; already-installed/downloaded copies elsewhere are not affected either way.

GET    /api/store/updates                                          List available updates for installed apps
     curl -s "$HALO_API_URL/api/store/updates" -H "Authorization: Bearer $HALO_API_TOKEN" | head -c 4000
     # out: {success:true,data:[{appId,currentVersion,latestVersion,entry,strategy,severity}]}

POST   /api/store/updates/<appId>/apply                            Apply an available update to an installed app
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/store/updates/<appId>/apply" \
          -H "Authorization: Bearer $HALO_API_TOKEN" \
          -H "Content-Type: application/json" \
          -d '{"mode":"force"}'
     # body: {"mode":"force"}
     # out: {success:true,data:{appId,from,to,severity,kept,editsKnown}} or {success:false,error}
     # note: mode is optional: "patch_minor"|"major"|"force", defaults to "force". No built-in downgrade path — reinstalling the old version is the way back.
     # note: A digital human keeps the user's version of every field that differs from the author's new version (prompt, run schedule, name, ...); kept lists those fields, empty when the update applied in full. editsKnown false means there was no earlier author version to compare with, so every difference was kept. The user can switch kept fields to the author's version in the app.

POST   /api/store/updates/check-now                                Re-check all installed apps for available updates right now
     # note: reversible — can be undone or corrected afterward.
     curl -sX POST "$HALO_API_URL/api/store/updates/check-now" -H "Authorization: Bearer $HALO_API_TOKEN"
     # out: {success:true,data:[{appId,currentVersion,latestVersion,entry,strategy,severity}]}
     # note: Apps set to auto-update silently apply any patch/minor update found — this is not just a read.
